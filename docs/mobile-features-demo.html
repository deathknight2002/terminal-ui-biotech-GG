🔗 API Endpoints

Backend routes for mobile features:

WebSocket Events

ws://localhost:3001
// Subscribe to streams socket.emit('subscribe', { streams: ['market:VRTX', 'biotech:updates'] }) // AI chat streaming socket.emit('ai_chat_stream', { message: 'Tell me about VRTX', messageId: 'msg-123' }) socket.on('ai_stream_chunk', (data) => console.log(data.chunk))

Notifications API

POST /api/notifications/register
POST /api/notifications/subscribe/fda
POST /api/notifications/subscribe/price
// Register device POST /api/notifications/register { "token": "device-token", "platform": "ios", "deviceId": "device-123" } // Subscribe to FDA alerts POST /api/notifications/subscribe/fda { "token": "device-token", "symbols": ["VRTX", "REGN"] } // Subscribe to price alerts (5% threshold) POST /api/notifications/subscribe/price { "token": "device-token", "symbols": ["VRTX"], "threshold": 5 }

Widgets API

GET /api/widgets/portfolio-summary
GET /api/widgets/news-headlines?limit=5
GET /api/widgets/catalyst-calendar?days=7
// Get portfolio summary GET /api/widgets/portfolio-summary → { totalValue, dayChange, dayChangePercent, topHoldings } // Get news headlines GET /api/widgets/news-headlines?limit=5 → [{ id, title, source, timestamp, category, sentiment }]
📚 Usage Examples

WebSocket Streaming

import { useWebSocket } from './hooks/useWebSocket'; const { isConnected, emit, on, subscribe } = useWebSocket({ url: 'http://localhost:3001', autoConnect: true, }); // Subscribe to data streams subscribe(['market:VRTX', 'biotech:updates']); // Listen for updates on('market_data', (data) => console.log('Market update:', data));

Voice Input

import { useSpeechRecognition } from './hooks/useSpeechRecognition'; const { isListening, transcript, start, stop } = useSpeechRecognition({ onResult: (text) => setInputValue(text), }); <button onClick={start}>{isListening ? '⏹️' : '🎤'}</button>

Offline Storage

import { offlineStorage, STORES } from './services/offlineStorage'; // Initialize await offlineStorage.initialize(); // Cache data await offlineStorage.cacheCompanies(companies); await offlineStorage.cachePipeline(pipeline); // Retrieve const cached = await offlineStorage.getCachedCompanies();

Advanced Charts

import { CandlestickChart } from './components/CandlestickChart'; <CandlestickChart data={priceData} title="VRTX Stock Price" showVolume={true} technicalIndicators={{ sma: [{ period: 20 }, { period: 50 }], ema: [{ period: 12 }], bollinger: { period: 20, stdDev: 2 }, }} />
